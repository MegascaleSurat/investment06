### Watchlist Management API cURL List

# 1. Get Watchlist & Uploads
curl -X GET http://localhost:5000/api/v1/watchlist \
     -H "Authorization: Bearer <ACCESS_TOKEN>"

# 2. Add Stock to Watchlist
curl -X POST http://localhost:5000/api/v1/watchlist/add \
     -H "Authorization: Bearer <ACCESS_TOKEN>" \
     -H "Content-Type: application/json" \
     -d '{
           "symbol": "RELIANCE",
           "exchange": "NSE",
           "entryPrice": 2500.50,
           "stopLoss": 2400.00,
           "target": 2700.00,
           "targetMode": "PRICE"
         }'

# 3. Update Tracked Stock (By Code symbol:exchange)
curl -X PUT http://localhost:5000/api/v1/watchlist/RELIANCE:NSE \
     -H "Authorization: Bearer <ACCESS_TOKEN>" \
     -H "Content-Type: application/json" \
     -d '{
           "entryPrice": 2510.00,
           "stopLoss": 2420.00
         }'

# 4. Remove Stock from Watchlist
curl -X DELETE http://localhost:5000/api/v1/watchlist/RELIANCE:NSE \
     -H "Authorization: Bearer <ACCESS_TOKEN>"

# 5. Upload Watchlist (CSV/JSON Simulation)
curl -X POST http://localhost:5000/api/v1/watchlist/upload \
     -H "Authorization: Bearer <ACCESS_TOKEN>" \
     -H "Content-Type: application/json" \
     -d '{
           "filename": "my_watchlist.json",
           "stocks": [
             {
               "symbol": "TCS",
               "exchange": "NSE",
               "entryPrice": 3200.00,
               "stopLoss": 3100.00,
               "target": 3500.00
             },
             {
               "symbol": "INFY",
               "exchange": "NSE",
               "entryPrice": 1500.00,
               "stopLoss": 1450.00,
               "target": 1650.00
             }
           ]
         }'
