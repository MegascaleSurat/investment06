### Authentication & Users API cURL List

# 1. Register a new user
curl -X POST http://localhost:5000/api/v1/auth/register \
     -H "Content-Type: application/json" \
     -d '{
           "fullName": "John Doe",
           "email": "john@example.com",
           "password": "SecurePassword123!",
           "phone": "+919876543210",
           "role": "USER"
         }'

# 2. Login
curl -X POST http://localhost:5000/api/v1/auth/login \
     -H "Content-Type: application/json" \
     -d '{
           "email": "john@example.com",
           "password": "SecurePassword123!"
         }'

# 3. Logout (Requires Access Token)
curl -X POST http://localhost:5000/api/v1/auth/logout \
     -H "Authorization: Bearer <ACCESS_TOKEN>"

# 4. Refresh Token
curl -X POST http://localhost:5000/api/v1/auth/refresh-token \
     -H "Content-Type: application/json" \
     -d '{
           "refreshToken": "<REFRESH_TOKEN>"
         }'

# 5. Get Profile (Requires Access Token)
curl -X GET http://localhost:5000/api/v1/auth/me \
     -H "Authorization: Bearer <ACCESS_TOKEN>"

# 6. Update Profile (Requires Access Token)
curl -X PUT http://localhost:5000/api/v1/auth/me \
     -H "Authorization: Bearer <ACCESS_TOKEN>" \
     -H "Content-Type: application/json" \
     -d '{
           "fullName": "John Updated",
           "phone": "+910000000000"
         }'

# 7. Change Password (Requires Access Token)
curl -X POST http://localhost:5000/api/v1/auth/change-password \
     -H "Authorization: Bearer <ACCESS_TOKEN>" \
     -H "Content-Type: application/json" \
     -d '{
           "oldPassword": "SecurePassword123!",
           "newPassword": "NewStrongPassword456!"
         }'
